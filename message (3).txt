--=============================================================
--  🔐 VALIDAÇÃO NA API
--=============================================================
local WEBHOOK_USUARIO = nil

local function validarKey()
    local req = syn and syn.request or http_request or request or http.request
    if not req then return false, "Executor sem suporte HTTP." end

    local hwid = tostring(LocalPlayer.UserId)
    
    local ok, res = pcall(function()
        return req({
            Url = API_URL,
            Method = "POST",
            Headers = { ["Content-Type"] = "application/json" },
            Body = HttpService:JSONEncode({key = API_KEY, hwid = hwid})
        })
    end)

    if ok and res and res.StatusCode == 200 then
        local data = HttpService:JSONDecode(res.Body)
        if data.sucesso then
            WEBHOOK_USUARIO = data.webhook
            return true, "OK"
        end
    end
    return false, "Key inválida."
end

local isKeyValid, motivo = validarKey()
if not isKeyValid then
    LocalPlayer:Kick("❌ Falha na validação: " .. motivo)
    return
end
